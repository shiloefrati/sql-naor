-- יצירת סיקוונסים
CREATE SEQUENCE CITY_ID_SEQ START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE SCHOOL_ID_SEQ START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE STUDENT_ID_SEQ START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE TEST_ID_SEQ START WITH 1 INCREMENT BY 1;

-- יצירת טבלת ערים
CREATE TABLE naorBony_cities (
    id INT DEFAULT CITY_ID_SEQ.NEXTVAL PRIMARY KEY,
    cityName VARCHAR2(168),
    location VARCHAR2(30)
);

-- יצירת טבלת בתי ספר
CREATE TABLE naorBony_schools (
    id INT DEFAULT SCHOOL_ID_SEQ.NEXTVAL PRIMARY KEY,
    schoolName VARCHAR2(40),
    cityId INT REFERENCES naorBony_cities(id)
);

-- יצירת טבלת תלמידים
CREATE TABLE naorBony_students (
    id INT DEFAULT STUDENT_ID_SEQ.NEXTVAL PRIMARY KEY,
    firstName VARCHAR2(255),
    lastName VARCHAR2(255),
    school INT REFERENCES naorBony_schools(id)
);

-- יצירת טבלת מבחנים
CREATE TABLE naorBony_tests (
    id INT DEFAULT TEST_ID_SEQ.NEXTVAL PRIMARY KEY,
    subject VARCHAR2(27),
    grade INT,
    testDate DATE,
    school INT REFERENCES naorBony_schools(id),
    student INT REFERENCES naorBony_students(id)
);
