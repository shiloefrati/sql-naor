SELECT students.firstName || ' ' || students.lastName AS fullName,
       AVG(grade) AS avg_grade
FROM naorBony_students students
INNER JOIN naorBony_tests tests ON students.id = tests.student 
GROUP BY students.firstName, students.lastName;

-- b
SELECT schools.schoolName AS school, AVG(grade)
FROM naorBony_schools schools
INNER JOIN naorBony_tests tests ON schools.id = tests.school 
GROUP BY schools.schoolName;

-- c
SELECT schools.schoolName AS schoolName, AVG(grade)
FROM naorBony_schools schools
INNER JOIN naorBony_tests tests ON schools.id = tests.school 
WHERE schools.schoolName = 'עירוני א' AND tests.subject = 'ביולוגיה'
GROUP BY schools.schoolName;

-- d
SELECT schools.schoolName AS school, AVG(grade)
FROM naorBony_schools schools
INNER JOIN naorBony_tests tests ON schools.id = tests.school 
WHERE tests.subject = 'כימיה'
GROUP BY schools.schoolName
ORDER BY AVG(grade) DESC;

-- e
SELECT schools.schoolName AS school, AVG(tests.grade) AS avg_grade
FROM naorBony_schools schools
LEFT JOIN naorBony_tests tests 
    ON schools.id = tests.school AND tests.subject = 'כימיה'
GROUP BY schools.schoolName
ORDER BY avg_grade DESC NULLS LAST;

-- f
SELECT AVG(grade) AS avg_grade , students.firstName || ' ' || students.lastName AS fullName
FROM naorBony_students students
INNER JOIN naorBony_tests tests ON students.id = tests.student 
GROUP BY students.firstName, students.lastName
ORDER BY avg_grade DESC
FETCH FIRST 1 ROWS ONLY;

-- g
SELECT AVG(grade) AS avg_grade , students.firstName || ' ' || students.lastName AS fullName
FROM naorBony_students students
INNER JOIN naorBony_tests tests ON students.id = tests.student
INNER JOIN naorBony_schools schools ON students.school = schools.id 
WHERE schools.schoolName = 'עירוני ב' AND tests.subject = 'מתמטיקה'
GROUP BY students.firstName, students.lastName
ORDER BY avg_grade DESC
FETCH FIRST 1 ROWS ONLY;

-- h
SELECT MAX(avg_grade) AS max_avg, schoolName
FROM (
    SELECT AVG(grade) AS avg_grade, 
           students.firstName || ' ' || students.lastName AS fullName,
	       schools.schoolName AS schoolName
	FROM naorBony_students students
	INNER JOIN naorBony_tests tests ON students.id = tests.student
	INNER JOIN naorBony_schools schools ON students.school = schools.id  
	GROUP BY students.firstName, students.lastName, schools.schoolName
	ORDER BY avg_grade DESC 
)
GROUP BY schoolName
ORDER BY max_avg ASC;

CREATE VIEW schoolFromNorth AS
SELECT schoolName, id
FROM naorBony_schools schools
INNER JOIN naorBony_cities cities ON schools.city = cities.id 
WHERE cities.location = 'צפון';

--f
CREATE VIEW studentByAvg AS
SELECT students.firstName || ' ' || students.lastName AS fullName,
       AVG(grade) AS avg_grade, 
       students.school
FROM naorBony_students students
INNER JOIN naorBony_tests tests ON students.id = tests.student 
GROUP BY students.firstName, students.lastName, students.school
HAVING AVG(grade) > 67;

--f
SELECT sba.fullName, sba.avg_grade
FROM studentByAvg sba
INNER JOIN schoolFromNorth sfn ON sba.school = sfn.id;
ORDER BY sba.avg_grade DESC
